# Earthsound v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A static website where people explore NASA Earth Information Center visualizations by ear (Jukebox) and play 3 hand-curated 5-round ear-training days whose answers come from NASA pixels, with reveals that sync the original EIC imagery to the sound.

**Architecture:**
- An offline Python pipeline decodes 3 NASA SVS layers (night lights, sea surface temperature, ISS lightning) into `site/data/rounds.json` (fully formed rounds with answers) and `site/data/grids.json` (2° value grids for the Jukebox), plus thumbnails.
- The browser side is vanilla ES modules: `game.js` (pure logic, tested with `node --test`), `sound.js` (Web Audio synthesis), `explore.js` (Jukebox), `app.js` (DOM wiring).
- No backend. Deployed to GitHub Pages.

**Tech Stack:** Python 3 + numpy + Pillow + pytest; HTML/CSS/vanilla JS + Web Audio API; Node 20+ test runner; GitHub Actions Pages.

**Spec:** `spaceapps-2026/specs/2026-11-14-earthsound-v2-spec.md`. Data calibrations come from `spaceapps-2026/notes/R2-technical-redteam.md` and `R3-guessing-recon.md`.

```python
"""lat/lon -> pixel for each NASA SVS asset. Calibrations measured in notes/R2-technical-redteam.md."""

def nl_px(lat, lon):
    # Night-lights 7680x2160 frame holds the world twice; use the left 3840x2160 copy.
    # Its left edge is lon -168.8 (not -180), stretched 360x180 deg -> 3840x2160 (R2, checked on Chicago + Tokyo).
    return ((lon + 168.8) % 360) / 360 * 3840, (90 - lat) / 180 * 2160

def eq_px(lat, lon, w, h):
    # Plain equirectangular, lon -180..180 (SST 3912, lightning 5011).
    return (lon + 180) / 360 * w, (90 - lat) / 180 * h

def window(a, cx, cy, rx, ry):
    return a[max(0, int(cy - ry)):int(cy + ry) + 1, max(0, int(cx - rx)):int(cx + rx) + 1]
```

```python
"""Decode NASA SVS pixels into values. Methods from notes/feasibility/scripts/r3_*.py."""
import colorsys
import numpy as np
from PIL import Image
from geo import nl_px, eq_px, window

def load(path):
    return np.asarray(Image.open(path).convert('RGB')).astype(float)

def city_lights(f12, f23, fch, lat, lon, r=1.5):
    """Light sums 2012 vs 2023 in a +-r deg window, plus NASA change-map sign and 2023 p99."""
    cx, cy = nl_px(lat, lon)
    rx, ry = r / 360 * 3840, r / 180 * 2160
    a = window(f12, cx, cy, rx, ry).mean(2)
    b = window(f23, cx, cy, rx, ry).mean(2)
    c = window(fch, cx, cy, rx, ry)
    bg = np.percentile(a, 20)
    l12 = float(np.clip(a - bg - 3, 0, None).sum())
    l23 = float(np.clip(b - bg - 3, 0, None).sum())
    purple_minus_orange = c[..., 2] - c[..., 0]          # change map: purple = new lights, orange = lost
    lit = c.max(2) > 45
    net = float(((purple_minus_orange > 10) & lit).mean() - ((purple_minus_orange < -10) & lit).mean())
    return {'l2012': l12, 'l2023': l23, 'rel': (l23 - l12) / (l12 + 1e-6),
            'changemap': net, 'p99': float(np.percentile(b, 99))}

def sst_decoder(cbar_path):
    """Returns decode(path) -> 2D array of deg C (NaN = land/no data), via colorbar hue (0-32 C)."""
    cb = load(cbar_path)[40, 14:292] / 255
    hues = np.array([colorsys.rgb_to_hsv(*c)[0] for c in cb])
    temps = np.linspace(0, 32, len(cb))
    order = np.argsort(hues)

    def decode(path):
        a = load(path) / 255
        mx, mn = a.max(2), a.min(2)
        sat = (mx - mn) / (mx + 1e-9)
        ok = (sat > 0.5) & (mx > 0.35)
        r, g, b = a[..., 0], a[..., 1], a[..., 2]
        d = mx - mn + 1e-9
        h = np.where(mx == r, ((g - b) / d) % 6, np.where(mx == g, (b - r) / d + 2, (r - g) / d + 4)) / 6
        T = np.interp(h, hues[order], temps[order])
        T[~ok] = np.nan
        return T
    return decode

def sst_at(T, lat, lon, r=3):
    h, w = T.shape
    cx, cy = eq_px(lat, lon, w, h)
    win = window(T, cx, cy, r / 360 * w, r / 180 * h)
    return float(np.nanmean(win)) if np.isfinite(win).sum() > 20 else None

def lightning_mask(path):
    a = load(path)
    return (a[..., 0] > 150) & (a[..., 1] > 130) & (a[..., 2] < 80)   # yellow event dots

def hemis(mask):
    half = mask.shape[0] // 2
    return int(mask[:half].sum()), int(mask[half:].sum())

def features(path):
    """Global image features for the 'which visualization made this sound' round (all 0..1)."""
    a = load(path) / 255
    mx, mn = a.max(2), a.min(2)
    sat = (mx - mn) / (mx + 1e-9)
    vivid = (sat > 0.3) & (mx > 0.3)
    hue = colorsys.rgb_to_hsv(*a[vivid].mean(0))[0] if vivid.any() else 0.0
    return {'bright': float(mx.mean()),
            'sat': float(sat[mx > 0.2].mean()) if (mx > 0.2).any() else 0.0,
            'hue': float(hue),
            'edges': float(min(1.0, np.abs(np.diff(mx, axis=1)).mean() * 10))}
```

```python
    lit_paths = sorted(glob.glob(os.path.join(RAW, 'lit', 'f_*.png')), key=lambda p: float(p.split('f_')[1][:-4]))

    imgs = [('Night lights', 'data/img/nl_2023.jpg', 'nl_2023.png'), ('Ocean temperature', 'data/img/sst_mar.jpg', 'sst_mar.tif'),
            ('Lightning', 'data/img/lightning.jpg', 'lit/f_63.5.png'), ('Carbon dioxide', 'data/img/co2.jpg', 'co2.png')]

            'reveal': f'That was {label.lower()}: brightness set the register, colour set the chord, detail set the pulse.'})

    # 2. then vs now: |change| >= 30% and NASA change map agrees on the sign
    for n in lit:
        d = L[n]
        if abs(d['rel']) >= 0.30 and np.sign(d['rel']) == np.sign(d['changemap']):
            up = d['rel'] > 0

    # 3a. which has more light (2023 p99 gap >= 60 of 255)
    pairs = [(a, b) for a, b in itertools.combinations(lit, 2) if abs(L[a]['p99'] - L[b]['p99']) >= 60]
    for a, b in cap(pairs, 120, rnd):

        win = a if L[a]['p99'] > L[b]['p99'] else b
        rounds['whichMore'].append({'type': 'whichMore', 'prompt': f'Night lights, 2023. Left: {a}. Right: {b}. Which is brighter?',

            rounds['whichMore'].append({'type': 'whichMore', 'prompt': f'Sea surface, March. Left: {a}. Right: {b}. Which is warmer?',
                'options': [a, b], 'answer': win,

    counts = [hemis(lightning_mask(p)) for p in lit_paths]
    peak = max(n + s for n, s in counts)
    for p, (n, s) in zip(lit_paths, counts):

    triples = [t for t in itertools.combinations(lit, 3) if all(abs(L[x]['rel'] - L[y]['rel']) >= 0.25 for x, y in itertools.combinations(t, 2))]
    for t in cap(triples, 120, rnd):
        t = list(t); rnd.shuffle(t)
        best = max(t, key=lambda c: L[c]['rel'])

        r['marks'] = [CITIES.get(n) or OCEANS[n] for n in r['options']]
        r['image'] = 'data/img/nl_2023.jpg' if r['options'][0] in CITIES else 'data/img/sst_mar.jpg'
    for kind, pool in rounds.items():
        for j, r in enumerate(pool):
            r['id'] = f'{kind}-{j}'
    with open(os.path.join(out_dir, 'rounds.json'), 'w') as fh:

    write_grids(out_dir, f23, T, lightning_mask(os.path.join(RAW, 'lit', 'f_63.5.png')))
    return out_dir
```

```js
// Web Audio synthesis for every round type. No libraries.
let ctx;
export function audio() {
  ctx ??= new (globalThis.AudioContext || globalThis.webkitAudioContext)();
  if (ctx.state === 'suspended') ctx.resume();   // iOS: must be called from a tap/keypress
  return ctx;
}

const PENTA = [0, 2, 4, 7, 9];
export function quantize(semis) {
  const oct = Math.floor(semis / 12), r = semis - oct * 12;
  return oct * 12 + PENTA.reduce((a, b) => (Math.abs(b - r) < Math.abs(a - r) ? b : a));
}
export const freqFromSemis = (s) => 220 * 2 ** (s / 12);
export const freqFromUnit = (v) => freqFromSemis(quantize(Math.round(Math.max(0, Math.min(1, v)) * 24)));

function note(c, t0, { freq, endFreq = freq, dur, pan = 0, type = 'triangle', gain = 0.2 }) {
  const o = c.createOscillator(), g = c.createGain(), p = c.createStereoPanner();
  const edge = Math.min(0.05, dur / 4);
  o.type = type;
  o.frequency.setValueAtTime(freq, t0);
  if (endFreq !== freq) o.frequency.exponentialRampToValueAtTime(endFreq, t0 + dur);
  g.gain.setValueAtTime(0, t0);
  g.gain.linearRampToValueAtTime(gain, t0 + edge);
  g.gain.setValueAtTime(gain, t0 + dur - edge);
  g.gain.linearRampToValueAtTime(0, t0 + dur);
  p.pan.value = pan;
  o.connect(g).connect(p).connect(c.destination);
  o.start(t0);
  o.stop(t0 + dur + 0.02);
}

function clicks(c, t0, { rate, freq, dur }) {
  for (let t = 0; rate > 0 && t < dur; t += 1 / rate) note(c, t0 + t, { freq, dur: 0.04, type: 'square', gain: 0.12 });
}

const PANS = { 1: [0], 2: [-0.7, 0.7], 3: [-0.7, 0, 0.7] };

// spec comes from rounds.json; returns the duration in seconds
export function playSound(spec) {
  const c = audio(), t0 = c.currentTime + 0.05;
  if (spec.kind === 'glides') {
    const base = 261.63;   // C4; the glide's size in semitones = 12*log2(2023/2012)
    spec.glides.forEach((s, k) => note(c, t0 + k * 2.2, { freq: base, endFreq: base * 2 ** (s / 12), dur: 1.8, pan: PANS[spec.glides.length][k] }));
    return spec.glides.length * 2.2;
  }
  if (spec.kind === 'pair') {
    spec.values.forEach((v, k) => note(c, t0 + k * 1.5, { freq: freqFromUnit(v), dur: 1.2, pan: PANS[2][k] }));
    return 3;
  }
  if (spec.kind === 'clicks') {
    clicks(c, t0, { rate: spec.north, freq: 1760, dur: 3 });
    clicks(c, t0, { rate: spec.south, freq: 220, dur: 3 });
    return 3;
  }
  if (spec.kind === 'chord') {
    const root = 110 * 2 ** Math.round(spec.register * 3) * 2 ** (spec.root / 12);
    const steps = [0, 4, 7, 12].slice(0, spec.voices);
    const dur = Math.min(0.8, 0.9 / spec.rate);
    for (let t = 0; t < 3; t += 1 / spec.rate) steps.forEach((s) => note(c, t0 + t, { freq: root * 2 ** (s / 12), dur, gain: 0.1 }));
    return 3;
  }
  throw new Error(`unknown sound kind ${spec.kind}`);
}
```

```js
// Pure game logic: no DOM, no audio. Tested with node --test.
// ponytail: 3 hand-curated days (days.json) instead of a date-seeded daily game; add a seed when it goes daily.
export function dayRounds(pools, days, day) {
  if (!days[day - 1]) throw new Error(`no day ${day}`);
  const byId = Object.fromEntries(Object.values(pools).flat().map((r) => [r.id, r]));
  return days[day - 1].map((id) => {
    if (!byId[id]) throw new Error(`unknown round id ${id}`);
    return byId[id];
  });
}

export function score(results) {
  let streak = 0, bestStreak = 0;
  for (const x of results) { streak = x.correct ? streak + 1 : 0; bestStreak = Math.max(bestStreak, streak); }
  return { correct: results.filter((x) => x.correct).length, total: results.length, bestStreak };
}

export function shareLine(results, label) {
  const s = score(results);
  return `Earthsound ${label} ${results.map((x) => (x.correct ? '🟢' : '⚪')).join('')} ${s.correct}/${s.total}`;
}

export function toCSV(sessions) {
  const rows = ['tester,phase,day,round,type,answer,correct,ms'];
  for (const s of sessions)
    for (const x of s.results)
      rows.push([s.tester, s.phase, s.day, x.round, x.type, JSON.stringify(x.answer), x.correct, x.ms].join(','));
  return rows.join('\n');
}
```

```js
import { dayRounds, shareLine, toCSV } from './game.js';
import { audio, playSound } from './sound.js';
import { initExplore } from './explore.js';

const $ = (id) => document.getElementById(id);
const params = new URLSearchParams(location.search);
const tester = params.get('tester'), phase = params.get('phase') || 'daily';
const day = Number(params.get('day')) || 1;
const reduceMotion = matchMedia('(prefers-reduced-motion: reduce)').matches;

export function say(text) {
  if ($('narrate').checked && 'speechSynthesis' in window) {
    speechSynthesis.cancel();
    speechSynthesis.speak(new SpeechSynthesisUtterance(text));
  }
}

function readLog() { try { return JSON.parse(localStorage.getItem('earthsound-log')) || []; } catch { return []; } }
function writeLog(log) { try { localStorage.setItem('earthsound-log', JSON.stringify(log)); } catch { /* private mode: ignore */ } }

// tabs
for (const [btn, sec, other, otherSec] of [['tab-play', 'play', 'tab-explore', 'explore'], ['tab-explore', 'explore', 'tab-play', 'play']]) {
  $(btn).onclick = () => { $(sec).hidden = false; $(otherSec).hidden = true; $(btn).setAttribute('aria-pressed', 'true'); $(other).setAttribute('aria-pressed', 'false'); };
}
// intro: the tap also unlocks audio on iOS
$('begin').onclick = () => {
  audio();
  $('intro').hidden = true;
  (tester || params.has('day') ? $('tab-play') : $('tab-explore')).click();   // test links open on the challenge
};

// tutorial demos
document.querySelectorAll('[data-demo]').forEach((b) => { b.onclick = () => playSound(JSON.parse(b.dataset.demo)); });

const [pools, days, stories] = await Promise.all([
  fetch('data/rounds.json').then((r) => r.json()),
  fetch('data/days.json').then((r) => r.json()),
  fetch('data/stories.json').then((r) => r.json()).catch(() => ({})),
]);
const rounds = dayRounds(pools, days, day);
const results = [];
let i = 0, shownAt = 0;

function show() {
  const r = rounds[i];
  $('progress').textContent = `Day ${day}, round ${i + 1} of ${rounds.length}`;
  $('stage').hidden = true;
  $('prompt').textContent = r.prompt;
  say(r.prompt);
  $('images').replaceChildren(...(r.images || []).map((src, k) => Object.assign(new Image(), { src, alt: r.options[k] })));
  $('options').replaceChildren(...r.options.map((o, k) => {
    const b = document.createElement('button');
    b.textContent = `${k + 1}. ${o}`;
    b.onclick = () => answer(o);
    return b;
  }));
  $('reveal').textContent = '';
  $('next').hidden = true;
  shownAt = performance.now();
}

function answer(choice) {
  if (!$('next').hidden) return;
  const r = rounds[i], correct = choice === r.answer;
  results.push({ round: i + 1, type: r.type, answer: choice, correct, ms: Math.round(performance.now() - shownAt) });
  const story = stories[r.city] || stories[r.answer] || '';
  $('reveal').textContent = `${correct ? 'Correct.' : `Not quite: ${r.answer}.`} ${r.reveal} ${story}`;
  say($('reveal').textContent);
  stage(r);
  $('options').querySelectorAll('button').forEach((b) => { b.disabled = true; });
  $('next').hidden = false;
  $('next').focus();
}

const loadImg = (src) => new Promise((ok, fail) => Object.assign(new Image(), { onload() { ok(this); }, onerror: fail, src }));

// Reveal: the original EIC frame, synced to the sound. thenNow crossfades 2012 -> 2023 while the glide replays.
async function stage(r) {
  const c = $('stage'), g = c.getContext('2d');
  c.setAttribute('aria-label', r.reveal);
  if (r.zoom) {
    const [before, after] = await Promise.all(r.zoom.map(loadImg));
    c.hidden = false;
    const dur = playSound(r.sound) * 1000, t0 = performance.now();
    const frame = (now) => {
      const k = reduceMotion ? 1 : Math.min(1, (now - t0) / dur);
      g.globalAlpha = 1; g.drawImage(before, 0, 0, c.width, c.height);
      g.globalAlpha = k; g.drawImage(after, 0, 0, c.width, c.height);
      g.globalAlpha = 1; g.fillStyle = '#ffd166'; g.font = 'bold 28px system-ui';
      g.fillText(`${r.city}  ${k < 1 ? '2012 → 2023' : '2023'}`, 16, 40);
      if (k < 1) requestAnimationFrame(frame);
    };
    requestAnimationFrame(frame);
    return;
  }
  if (!r.image) return;   // match rounds already show their 4 images
  const im = await loadImg(r.image);
  c.hidden = false;
  g.drawImage(im, 0, 0, c.width, c.height);
  g.strokeStyle = '#ffd166'; g.fillStyle = '#ffd166'; g.lineWidth = 3; g.font = 'bold 18px system-ui';
  (r.marks || []).forEach(([lat, lon], k) => {   // all map images are equirectangular, lon -180..180
    const x = ((lon + 180) / 360) * c.width, y = ((90 - lat) / 180) * c.height;
    g.beginPath(); g.arc(x, y, 14, 0, 7); g.stroke();
    g.fillText(r.options[k], x + 18, y + 6);
  });
}

function end() {
  const log = readLog();
  log.push({ tester: tester || '', phase, day, results });
  writeLog(log);
  $('share').textContent = shareLine(results, `Day ${day}`);
  $('end').hidden = false;
  $('prompt').textContent = `Done. ${$('share').textContent}`;
  $('copy').focus();
}

$('start').onclick = () => playSound(rounds[i].sound);
$('next').onclick = () => { i += 1; i < rounds.length ? show() : end(); };
$('copy').onclick = () => navigator.clipboard?.writeText($('share').textContent);
$('export').onclick = () => {
  const url = URL.createObjectURL(new Blob([toCSV(readLog())], { type: 'text/csv' }));
  Object.assign(document.createElement('a'), { href: url, download: 'earthsound-results.csv' }).click();
};
document.addEventListener('keydown', (e) => {
  if ($('play').hidden || i >= rounds.length) return;
  if (e.key === ' ' && e.target.tagName !== 'BUTTON' && e.target.tagName !== 'SUMMARY') { e.preventDefault(); playSound(rounds[i].sound); }
  const n = Number(e.key);
  if (n >= 1 && n <= rounds[i].options.length) answer(rounds[i].options[n - 1]);
});

show();
initExplore(say);
```

```js
import { audio, freqFromUnit } from './sound.js';

const $ = (id) => document.getElementById(id);

export async function initExplore(say) {
  const { step, layers } = await fetch('data/grids.json').then((r) => r.json());
  const canvas = $('map'), g = canvas.getContext('2d');
  let layer = layers[0], img = new Image(), lat = 41, lon = -87, osc = null, gain = null;

  $('layer').replaceChildren(...layers.map((l, k) => Object.assign(document.createElement('option'), { value: k, textContent: l.label })));
  $('layer').onchange = () => { layer = layers[$('layer').value]; load(); };

  function value() {
    const row = Math.min(layer.values.length - 1, Math.floor((90 - lat) / step));
    const col = Math.min(layer.values[0].length - 1, Math.floor((lon + 180) / step));
    return layer.values[row][col];
  }
  function words(v) {
    if (v === null) return 'land, no ocean data';
    if (layer.id === 'sst') return `${v.toFixed(1)} degrees C`;
    if (layer.id === 'lit') return `${v} flashes`;
    return `${Math.round(v * 100)} percent brightness`;
  }
  function place() {
    return `${Math.abs(lat)}°${lat >= 0 ? 'N' : 'S'} ${Math.abs(lon)}°${lon >= 0 ? 'E' : 'W'}`;
  }
  function draw() {
    g.drawImage(img, 0, 0, canvas.width, canvas.height);
    const x = ((lon + 180) / 360) * canvas.width, y = ((90 - lat) / 180) * canvas.height;
    g.strokeStyle = '#ffd166'; g.lineWidth = 3;
    g.beginPath(); g.arc(x, y, 20, 0, 7); g.moveTo(x - 18, y); g.lineTo(x + 18, y); g.moveTo(x, y - 18); g.lineTo(x, y + 18); g.stroke();
  }
  function update() {
    draw();
    const v = value();
    $('ex-readout').textContent = `${place()}: ${words(v)}`;
    if (osc) {
      const u = v === null ? null : (v - layer.min) / (layer.max - layer.min);
      gain.gain.setTargetAtTime(u === null ? 0 : 0.15, audio().currentTime, 0.05);
      if (u !== null) osc.frequency.setTargetAtTime(freqFromUnit(u), audio().currentTime, 0.05);
    }
  }
  function move(dlon, dlat) {
    lon = ((lon + dlon + 540) % 360) - 180;
    lat = Math.max(-89, Math.min(89, lat + dlat));
    update();
  }
  function load() { img = new Image(); img.onload = update; img.src = layer.image; }

  $('ex-play').onclick = () => {
    const c = audio();
    if (osc) { osc.stop(); osc = null; return; }
    osc = c.createOscillator(); gain = c.createGain();
    osc.type = 'triangle'; gain.gain.value = 0;
    osc.connect(gain).connect(c.destination); osc.start();
    update();
  };
  $('ex-say').onclick = () => say($('ex-readout').textContent);
  document.querySelectorAll('[data-move]').forEach((b) => {
    const [dlon, dlat] = b.dataset.move.split(',').map(Number);
    b.onclick = () => move(dlon, dlat);
  });
  document.addEventListener('keydown', (e) => {
    if ($('explore').hidden || e.target.tagName === 'SELECT') return;
    const d = { ArrowLeft: [-step, 0], ArrowRight: [step, 0], ArrowUp: [0, step], ArrowDown: [0, -step] }[e.key];
    if (d) { e.preventDefault(); move(...d); }
    if (e.key === 'v' || e.key === 'V') say($('ex-readout').textContent);
  });

  // Phone tilt: tilt left/right moves west/east, tilt toward/away moves north/south.
  // ponytail: fixed 30 deg "neutral" holding angle; add a calibrate button if testers hold phones differently.
  $('tilt').onclick = async () => {
    if (typeof DeviceOrientationEvent?.requestPermission === 'function') {
      if ((await DeviceOrientationEvent.requestPermission()) !== 'granted') { $('ex-readout').textContent = 'Tilt permission denied.'; return; }
    }
    let beta = 30, gamma = 0;
    addEventListener('deviceorientation', (e) => { beta = e.beta ?? 30; gamma = e.gamma ?? 0; });
    setInterval(() => {
      const dlon = Math.round(gamma / 15), dlat = Math.round((30 - beta) / 15);
      if (dlon || dlat) move(dlon, dlat);
    }, 250);
    $('tilt').disabled = true;
    $('ex-readout').textContent = 'Tilt on. Tilt the phone to move.';
  };

  load();
}
```
